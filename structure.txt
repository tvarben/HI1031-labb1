CREATE DATABASE webshop_db;
CREATE USER 'webshop_user'@'localhost' IDENTIFIED BY '1234';
GRANT ALL PRIVILEGES ON webshop_db.* TO 'webshop_user'@'localhost';
FLUSH PRIVILEGES;

USE webshop_db;

-- Remove existing tables in dependency order
SET FOREIGN_KEY_CHECKS = 0;

DROP TABLE IF EXISTS OrderedItems;
DROP TABLE IF EXISTS Orders;
DROP TABLE IF EXISTS Items;
DROP TABLE IF EXISTS Users;

SET FOREIGN_KEY_CHECKS = 1;

CREATE TABLE User (
                      id INT AUTO_INCREMENT PRIMARY KEY,
                      username VARCHAR(100) NOT NULL,
                      password VARCHAR(255) NOT NULL,
                      role ENUM('customer', 'admin', 'employee') NOT NULL DEFAULT 'customer'
);


CREATE TABLE Item (
                      id INT AUTO_INCREMENT PRIMARY KEY,
                      name VARCHAR(255) NOT NULL,
                      price INT NOT NULL CHECK (price > 0),
                      quantity INT NOT NULL CHECK (quantity >= 0)
);


CREATE TABLE Orders (
                         id INT AUTO_INCREMENT PRIMARY KEY,
                         user_id INT NOT NULL,
                         created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                         total INT NOT NULL CHECK (total > 0),

                         FOREIGN KEY (user_id) REFERENCES User(id)
);


CREATE TABLE OrderedItem (
                             id INT AUTO_INCREMENT PRIMARY KEY,
                             order_id INT NOT NULL,
                             item_id INT NOT NULL,
                             quantity INT NOT NULL CHECK (quantity > 0),
                             price INT NOT NULL CHECK (price > 0),

                             FOREIGN KEY (order_id) REFERENCES Orders(id),
                             FOREIGN KEY (item_id) REFERENCES Item(id)
);
